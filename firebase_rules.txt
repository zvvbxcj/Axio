rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null && request.auth.uid == "JeKrAxN6u1Mbx21FHxZa5gXLBQ43";
    }

    function isAuth() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isAuth() && request.auth.uid == userId;
    }

    match /axioUsers/{userId} {
      allow read: if isAdmin() || isOwner(userId);
      allow create: if isOwner(userId)
        && !request.resource.data.keys().hasAny(['isBanned', 'isPremium', 'adminNotes', 'isShadowBanned']);
      allow update: if isAdmin() || (isOwner(userId)
        && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['isBanned', 'isPremium', 'adminNotes', 'isShadowBanned'])
        && request.resource.data.get('xp', 0) <= resource.data.get('xp', 0) + 2000
        && request.resource.data.get('level', 1) <= resource.data.get('level', 1) + 5
      );

      allow delete: if isAdmin();
    }

    match /leaderboard_stats/{userId} {
      allow read: if true;
      allow create: if isAdmin() || (isOwner(userId)
        && request.resource.data.keys().hasOnly([
          'name', 'avatar', 'xp', 'level', 'cookedDishes', 'co2Saved',
          'country', 'region', 'city'
        ])
        && request.resource.data.get('xp', 0) <= 2000
        && request.resource.data.get('level', 1) <= 5
      );

      allow update: if isAdmin() || (isOwner(userId)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'name', 'avatar', 'xp', 'level', 'cookedDishes', 'co2Saved',
          'country', 'region', 'city'
        ])
        && request.resource.data.get('xp', 0) <= resource.data.get('xp', 0) + 2000
        && request.resource.data.get('level', 1) <= resource.data.get('level', 1) + 5
      );

      allow delete: if isAdmin() || isOwner(userId);
    }

    match /global_settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /fridges/{fridgeId} {
      allow read: if isAdmin() || (isAuth() && request.auth.uid in resource.data.members);
      allow create: if isAuth()
        && request.resource.data.owner == request.auth.uid
        && request.auth.uid in request.resource.data.members;

      allow update: if isAdmin() || (isAuth() && (
        request.auth.uid in resource.data.members
        ||
        (
          !(request.auth.uid in resource.data.members) &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members']) &&
          request.resource.data.members == resource.data.members.concat([request.auth.uid])
        )
      ));

      allow delete: if isAdmin() || (isAuth() && resource.data.owner == request.auth.uid);
    }

    match /fridge_invites/{code} {
      allow read: if isAuth();
      allow create: if isAdmin() || (isAuth() && request.resource.data.owner == request.auth.uid);
      allow update: if isAdmin() || (isAuth() && resource.data.owner == request.auth.uid);
      allow delete: if isAdmin() || (isAuth() && resource.data.owner == request.auth.uid);
    }

    match /products/{productId} {
      allow create: if isAdmin() || (isAuth() &&
        request.auth.uid in get(/databases/$(database)/documents/fridges/$(request.resource.data.fridgeId)).data.members
      );

      allow read, update, delete: if isAdmin() || (isAuth() &&
        request.auth.uid in get(/databases/$(database)/documents/fridges/$(resource.data.fridgeId)).data.members
      );
    }

    match /public_recipes/{recipeId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /pending_recipes/{recipeId} {
      allow create: if isAuth() && request.resource.data.authorId == request.auth.uid;
      allow read: if isAdmin() || resource.data.authorId == request.auth.uid;
      allow update, delete: if isAdmin() || resource.data.authorId == request.auth.uid;
    }

    match /rejected_recipes/{recipeId} {
      allow read: if isAdmin() || resource.data.authorId == request.auth.uid;
      allow write: if isAdmin();
    }

    match /app_feedback/{docId} {
      allow create: if isAuth();
      allow read, update, delete: if isAdmin();
    }

    match /announcements/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /activity_log/{docId} {
      allow read, write: if isAuth() && request.auth.uid == resource.data.uid;
      allow create: if isAuth() && request.resource.data.uid == request.auth.uid; 
    }

    match /{document=**} {
      allow read, write: if isAdmin();
    }
  }
}
